# The streams of the conformance run (.github/workflows/build.yml, job "conformance"; docs/warp-decoder.md).
# testing/conformance/prepare_streams.sh reads this file, fetches or makes every stream and its md5
# reference, and writes the task list the decoder runs. Nothing listed here is stored in the repository.
#
#   aom <file>
#       A decoder test vector of the Alliance for Open Media, with the .md5 file published next to it:
#       https://storage.googleapis.com/aom-test-data/<file> and <file>.md5. All av1-1-b8-* and av1-1-b10-*
#       vectors of that bucket that have an .md5 file.
#   cut <name> <start seconds> <frames> <path>
#       The first <frames> frames from the key frame at or before <start seconds> of a Netflix Open Content
#       stream, http://download.opencontent.netflix.com.s3.amazonaws.com/<path>, copied without re-encoding
#       (ffmpeg -c copy); the reference is dav1d's output (testing/md5ref/make_md5_ref.sh).
#   svt <name> <frames> <svtav1-params>
#       A stream encoded at run time: ffmpeg's testsrc2 pattern, 1920x1080, 30 fps, 10-bit, libsvtav1
#       -preset 10 -crf 35 with the given -svtav1-params; the reference is dav1d's output.
#
# <name> and <file> are what the results and testing/conformance/expected.txt call the stream.

# 8-bit 3840x2160, one tile, frames that are decoded but not shown.
cut nf-dvbdash-a1-8b-2160p.ivf 0 20 AV1/DVB-DASH/A_1.mp4
# 8-bit 1280x720 and 480x270, global motion.
cut nf-chimera-8b-720p.ivf 60 40 AV1/Chimera/Old/Chimera-AV1-8bit-1280x720-3363kbps.mp4
cut nf-chimera-8b-270p.ivf 60 120 AV1/Chimera/Old/Chimera-AV1-8bit-480x270-552kbps.mp4
# 10-bit 1920x1080.
cut nf-chimera-10b-1080p.ivf 60 30 AV1/Chimera/Old/Chimera-AV1-10bit-1920x1080-6191kbps.mp4
cut nf-sparks-10b-1080p.ivf 40 40 AV1/Sparks/Sparks-5994fps-AV1-10bit-1920x1080-2194kbps.mp4

# Reference scaling: frames coded at other sizes than the frames they predict from.
svt syn-resize-10b-1080p.ivf 64 resize-mode=2
# Super-resolution: frames coded at a smaller width and upscaled, all denominators.
svt syn-superres-10b-1080p.ivf 64 superres-mode=2
# Eight tiles (four columns, two rows) and a key frame every 16 frames: the other streams have one tile
# and one key frame. Also the stream that testing/conformance/cut_tile_check.py damages for the decode
# error of a tile whose size does not fit, and decodes on from its next key frame.
svt syn-tiles-10b-1080p.ivf 40 tile-columns=2:tile-rows=1:keyint=16
